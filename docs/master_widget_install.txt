    "repo_password": "CHANGE_THIS_PASSWORD",
    "total_capacity_gb": 1000,
    "master_allocation_pct": 0.45,
    "backup_paths": {
        "active_work": ["/home/$USER/work/current-project"],
        "critical": ["/home/$USER/work", "/home/$USER/projects"],
        "important": ["/home/$USER/documents"],
        "standard": ["/home/$USER"]
    },
    "frequencies": {
        "active_work": 900,
        "critical": 3600,
        "important": 21600,
        "standard": 86400
    }
}

# Create automatic snapshot daemon script
cat > "$INSTALL_DIR/snapshot_daemon.py" << 'DAEMON_SCRIPT'
#!/usr/bin/env python3
"""
Background daemon for automatic snapshots
Runs continuously and creates snapshots based on schedule
"""

import time
import json
import logging
from pathlib import Path
from datetime import datetime

# Setup logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler(Path.home() / '.local/share/backup-system/daemon.log'),
        logging.StreamHandler()
    ]
)

# Load configuration
config_path = Path.home() / '.config/backup-system/config.json'
with open(config_path) as f:
    CONFIG = json.load(f)

# Import the backup manager (would need to be properly packaged)
import sys
sys.path.insert(0, str(Path(__file__).parent))

class SnapshotDaemon:
    """Daemon that runs automatic snapshots"""
    
    def __init__(self):
        self.last_snapshot = {}
        for category in CONFIG['backup_paths'].keys():
            self.last_snapshot[category] = 0
    
    def should_snapshot(self, category: str) -> bool:
        """Check if it's time for a snapshot"""
        now = time.time()
        interval = CONFIG['frequencies'][category]
        last = self.last_snapshot[category]
        
        return (now - last) >= interval
    
    def create_snapshot(self, category: str):
        """Create snapshot for category"""
        paths = CONFIG['backup_paths'][category]
        logging.info(f"Creating {category} snapshot for {paths}")
        
        # Run restic backup
        import subprocess
        cmd = [
            'restic', 'backup',
            '--tag', 'incremental',
            '--tag', category,
            '--tag', f"auto-{datetime.now().strftime('%Y%m%d-%H%M')}"
        ]
        cmd.extend(paths)
        
        env = {
            'RESTIC_REPOSITORY': CONFIG['repo_path'],
            'RESTIC_PASSWORD': CONFIG['repo_password']
        }
        
        try:
            result = subprocess.run(
                cmd,
                env=env,
                capture_output=True,
                text=True,
                timeout=600
            )
            
            if result.returncode == 0:
                logging.info(f"✓ {category} snapshot completed")
                self.last_snapshot[category] = time.time()
            else:
                logging.error(f"✗ {category} snapshot failed: {result.stderr}")
        except Exception as e:
            logging.error(f"✗ {category} snapshot error: {e}")
    
    def check_storage(self):
        """Check storage and prune if needed"""
        import subprocess
        
        env = {
            'RESTIC_REPOSITORY': CONFIG['repo_path'],
            'RESTIC_PASSWORD': CONFIG['repo_password']
        }
        
        result = subprocess.run(
            ['restic', 'stats', '--json'],
            env=env,
            capture_output=True,
            text=True
        )
        
        if result.returncode == 0:
            stats = json.loads(result.stdout)
            total_size = stats.get('total_size', 0)
            capacity = CONFIG['total_capacity_gb'] * 1e9
            usage_pct = (total_size / capacity) * 100
            
            logging.info(f"Storage: {usage_pct:.1f}% used")
            
            if usage_pct > 85:
                logging.warning("Storage critical, pruning...")
                subprocess.run([
                    'restic', 'forget',
                    '--tag', 'incremental',
                    '--keep-hourly', '24',
                    '--keep-daily', '7',
                    '--keep-weekly', '4'
                ], env=env, capture_output=True)
                
                subprocess.run(['restic', 'prune'], env=env, capture_output=True)
    
    def run(self):
        """Main daemon loop"""
        logging.info("Snapshot daemon started")
        
        iteration = 0
        while True:
            try:
                # Check each category
                for category in CONFIG['backup_paths'].keys():
                    if self.should_snapshot(category):
                        self.create_snapshot(category)
                
                # Check storage every 10 iterations (~10 minutes)
                iteration += 1
                if iteration % 10 == 0:
                    self.check_storage()
                
                time.sleep(60)  # Check every minute
                
            except KeyboardInterrupt:
                logging.info("Daemon stopped by user")
                break
            except Exception as e:
                logging.error(f"Daemon error: {e}")
                time.sleep(60)

if __name__ == '__main__':
    daemon = SnapshotDaemon()
    daemon.run()
DAEMON_SCRIPT

chmod +x "$INSTALL_DIR/snapshot_daemon.py"

# Create systemd service for automatic snapshots
if [[ $EUID -eq 0 ]]; then
    SERVICE_FILE="$SERVICE_DIR/backup-snapshot.service"
else
    SERVICE_FILE="$SERVICE_DIR/backup-snapshot.service"
fi

cat > "$SERVICE_FILE" << SERVICE

Description=Rotating Master Image Backup Snapshot Daemon
After=network.target

[Service]
Type=simple
ExecStart=$INSTALL_DIR/snapshot_daemon.py
Restart=always
RestartSec=10

[Install]
WantedBy=default.target
SERVICE
