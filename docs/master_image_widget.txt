def init_repository(self):
        """Initialize restic repository"""
        reply = QMessageBox.question(
            self,
            "Initialize Repository",
            f"Initialize restic repository at {CONFIG['repo_path']}?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
        )
        
        if reply == QMessageBox.StandardButton.Yes:
            self.status_label.setText("Initializing repository...")
            QApplication.processEvents()
            
            success, output = self.manager.init_repo()
            
            if success:
                QMessageBox.information(self, "Success", "Repository initialized!")
            else:
                QMessageBox.warning(self, "Error", f"Initialization failed:\n{output}")
            
            self.refresh_stats()
