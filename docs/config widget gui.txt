class ScanWorker(QThread):
    """Background thread for scanning"""
    progress = pyqtSignal(str)
    finished = pyqtSignal(list, dict)
    
    def __init__(self, scanner):
        super().__init__()
        self.scanner = scanner
    
    def run(self):
        def progress_callback(msg):
            self.progress.emit(msg)
        
        results = self.scanner.scan(progress_callback)
        summary = self.scanner.get_summary()
        self.finished.emit(results, summary)

    def populate_results_tree(self, results: list):
        """Populate results tree"""
        self.results_tree.clear()
        
        for item in results:
            tree_item = QTreeWidgetItem()
            
            # Checkbox
            tree_item.setCheckState(0, Qt.CheckState.Checked if item['recommend_import'] else Qt.CheckState.Unchecked)
            
            # Icon and name
            icon = "🔒" if item['sensitive'] else "📄"
            tree_item.setText(1, f"{icon} {item['path'].name}")
            tree_item.setData(1, Qt.ItemDataRole.UserRole, item)
            
            # Category
            tree_item.setText(2, item['category'])
            
            # Size
            size_kb = item['size'] / 1024
            tree_item.setText(3, f"{size_kb:.1f} KB")
            
            # Modified
            if item['modified']:
                tree_item.setText(4, item['modified'].strftime('%Y-%m-%d %H:%M'))
            
            # Color code by importance
            if item['sensitive']:
                tree_item.setForeground(1, QColor(255, 100, 100))
            elif item['importance'] >= 4:
                tree_item.setForeground(1, QColor(100, 200, 100))
            
            self.results_tree.addTopLevelItem(tree_item)
    
    def filter_results(self):
        """Filter displayed results"""
        filter_text = self.filter_input.text().lower()
        show_recommended = self.show_recommended_only.isChecked()
        
        for i in range(self.results_tree.topLevelItemCount()):
            item = self.results_tree.topLevelItem(i)
            data = item.data(1, Qt.ItemDataRole.UserRole)
            
            # Check filter
            text_match = filter_text in str(data['path']).lower() if filter_text else True
            rec_match = data['recommend_import'] if show_recommended else True
            
            item.setHidden(not (text_match and rec_match))
    
    def show_config_details(self):
        """Show details of selected config"""
        selected = self.results_tree.selectedItems()
        if not selected:
            return
        
        item = selected[0]
        data = item.data(1, Qt.ItemDataRole.UserRole)
        
        details = f"""
Path: {data['path']}
Category: {data['category']}
Importance: {data['importance']}/5
Sensitive: {'Yes' if data['sensitive'] else 'No'}
Size: {data['size']/1024:.1f} KB
Modified: {data['modified'].strftime('%Y-%m-%d %H:%M:%S') if data['modified'] else 'Unknown'}
Recommended: {'Yes' if data['recommend_import'] else 'No'}
        """
        
        self.details_text.setText(details.strip())
        
        # Load preview
        try:
            with open(data['path'], 'r') as f:
                preview = f.read(500)  # First 500 chars
                if len(preview) == 500:
                    preview += "\n\n... (truncated)"
                self.preview_text.setText(preview)
        except:
            self.preview_text.setText("(Unable to preview)")
    
    def import_selected(self):
        """Import selected configs"""
        selected_count = 0
        for i in range(self.results_tree.topLevelItemCount()):
            item = self.results_tree.topLevelItem(i)
            if item.checkState(0) == Qt.CheckState.Checked:
                selected_count += 1
        
        if selected_count == 0:
            QMessageBox.warning(self, "No Selection", "No configs selected for import.")
            return
        
        reply = QMessageBox.question(
            self,
            "Confirm Import",
            f"Import {selected_count} selected configs to master directory?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
        )
        
        if reply == QMessageBox.StandardButton.Yes:
            self.perform_import(selected_only=True)
    
    def import_recommended(self):
        """Import all recommended configs"""
        recommended_count = sum(1 for r in self.scan_results if r['recommend_import'])
        
        reply = QMessageBox.question(
            self,
            "Confirm Import",
            f"Import {recommended_count} recommended configs?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
        )
        
        if reply == QMessageBox.StandardButton.Yes:
            self.perform_import(selected_only=False)
